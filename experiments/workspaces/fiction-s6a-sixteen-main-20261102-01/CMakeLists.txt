cmake_minimum_required(VERSION 3.20)
project(s6a_sixteen_main CXX)
set(CMAKE_CXX_STANDARD 20)
set(ROOT "${CMAKE_CURRENT_SOURCE_DIR}")
while(NOT EXISTS "${ROOT}/fiction.project.json")
 get_filename_component(PARENT "${ROOT}" DIRECTORY)
 if(PARENT STREQUAL ROOT)
  message(FATAL_ERROR "fiction.project.json not found")
 endif()
 set(ROOT "${PARENT}")
endwhile()
file(READ "${ROOT}/fiction.project.json" LAYOUT)
string(JSON SIM GET "${LAYOUT}" simulator)
add_subdirectory("${ROOT}/${SIM}/cpp" production)
add_executable(s6a_sixteen_main s6a_sixteen_main.cpp)
target_include_directories(s6a_sixteen_main PRIVATE "${ROOT}/${SIM}/cpp/include" "${CMAKE_CURRENT_SOURCE_DIR}/include")
target_link_libraries(s6a_sixteen_main PRIVATE fiction_core)
