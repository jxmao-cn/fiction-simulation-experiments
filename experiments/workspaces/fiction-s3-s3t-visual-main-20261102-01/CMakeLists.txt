cmake_minimum_required(VERSION 3.20)
project(s3_visual CXX)
set(CMAKE_CXX_STANDARD 20)
set(ROOT "${CMAKE_CURRENT_SOURCE_DIR}")
while(NOT EXISTS "${ROOT}/fiction.project.json")
 get_filename_component(PARENT "${ROOT}" DIRECTORY)
 if(PARENT STREQUAL ROOT)
  message(FATAL_ERROR "fiction.project.json not found")
 endif()
 set(ROOT "${PARENT}")
endwhile()
file(READ "${ROOT}/fiction.project.json" LAYOUT)
string(JSON SIM GET "${LAYOUT}" simulator)
add_subdirectory("${ROOT}/${SIM}/cpp" production)
find_package(ZLIB REQUIRED)
add_executable(dynamic_bar_visual dynamic_bar_visual.cpp)
target_link_libraries(dynamic_bar_visual PRIVATE fiction_core ZLIB::ZLIB)
target_compile_definitions(dynamic_bar_visual PRIVATE FICTION_HAS_ZLIB=1)
